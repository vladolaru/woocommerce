```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import hashlib
import hmac
import json
import os
import re
import sys
from pathlib import Path
from typing import Any


FLOW = "SC-02-blocks-card-checkout"
STATUS_EXIT = {"pass": 0, "fail": 1, "blocked": 3}
KEY_ENV = "CRITICAL_FLOWS_RUN_CONTEXT_KEY"
KEY_RE = re.compile(r"[0-9a-f]{64}")
DIGEST_RE = re.compile(r"sha256:[0-9a-f]{64}")
HMAC_RE = re.compile(r"hmac-sha256:[0-9a-f]{64}")
DOMAINS = {
    "state": b"woopayments-sc02-state-v1\0",
    "http": b"woopayments-sc02-http-v1\0",
    "result": b"woopayments-sc02-result-v1\0",
    "comparison": b"woopayments-sc02-comparison-v1\0",
    "execution": b"woopayments-sc02-execution-v1\0",
    "manifest": b"woopayments-sc02-manifest-v1\0",
}


def canonical(value: Any) -> bytes:
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")


def context_key() -> bytes | None:
    raw = os.environ.get(KEY_ENV, "")
    return bytes.fromhex(raw) if KEY_RE.fullmatch(raw) else None


def payload_digest(payload: dict[str, Any]) -> str:
    unsigned = dict(payload)
    unsigned.pop("payload_sha256", None)
    return "sha256:" + hashlib.sha256(canonical(unsigned)).hexdigest()


def context_hmac(kind: str, payload: dict[str, Any], key: bytes) -> str:
    unsigned = dict(payload)
    unsigned.pop("payload_sha256", None)
    unsigned.pop("context_hmac", None)
    return "hmac-sha256:" + hmac.new(key, DOMAINS[kind] + canonical(unsigned), hashlib.sha256).hexdigest()


def seal(kind: str, payload: dict[str, Any], key: bytes) -> dict[str, Any]:
    sealed = dict(payload)
    sealed["context_hmac"] = context_hmac(kind, sealed, key)
    sealed["payload_sha256"] = payload_digest(sealed)
    return sealed


def load_unique_json(raw: str) -> Any:
    def reject_duplicate(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
        result: dict[str, Any] = {}
        for name, value in pairs:
            if name in result:
                raise ValueError("duplicate JSON key")
            result[name] = value
        return result
    return json.loads(raw, object_pairs_hook=reject_duplicate)
```

```python
class StoreApiHandler(BaseHTTPRequestHandler):
    tokens = [f"fixture-cart-token-{index}-not-for-output" for index in range(6)]
    response_index = 0
    cart_ready = False
    requests: list[dict] = []

    def log_message(self, format: str, *args: object) -> None:
        return

    def _send(self, status: int, payload: dict, *, cart_token: bool = True) -> None:
        body = json.dumps(payload).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        if cart_token:
            self.send_header("Cart-Token", type(self).tokens[type(self).response_index])
            type(self).response_index += 1
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:
        self.requests.append({"method": "GET", "path": self.path, "cart_token": self.headers.get("Cart-Token")})
        self._send(200, final_cart() if type(self).cart_ready else empty_cart())

    def do_POST(self) -> None:
        length = int(self.headers.get("Content-Length", "0"))
        payload = json.loads(self.rfile.read(length) or b"{}")
        self.requests.append({"method": "POST", "path": self.path, "cart_token": self.headers.get("Cart-Token"), "payload": payload})
        type(self).cart_ready = True
        if self.path.endswith("/checkout"):
            self._send(200, {"order_id": 2167, "status": "processing", "payment_result": {"payment_status": "success", "payment_details": [], "redirect_url": "http://localhost/order-received/2167/"}})
        else:
            self._send(200, final_cart())
```

Use `urllib.request` only. The client must disable automatic redirect acceptance and manually reject a redirect whose resolved origin differs from the configured origin:

```python
class RefuseRedirects(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def allowed_base_url(raw: str) -> tuple[str, str]:
    parsed = urllib.parse.urlsplit(raw.rstrip("/"))
    host = (parsed.hostname or "").lower()
    if parsed.scheme not in {"http", "https"}:
        raise Blocked("Store API base URL must use HTTP or HTTPS.")
    if host not in {"localhost", "127.0.0.1"} and not host.endswith(".localhost"):
        raise Blocked("Store API base URL must stay local.")
    if parsed.username or parsed.password or parsed.query or parsed.fragment:
        raise Blocked("Store API base URL contains forbidden URL components.")
    origin = f"{parsed.scheme}://{parsed.netloc}"
    return origin, raw.rstrip("/")


def request_json(client, method: str, path: str, body: dict | None, cart_token: str | None) -> tuple[int, dict, str | None]:
    headers = {"Accept": "application/json"}
    encoded = None
    if body is not None:
        headers["Content-Type"] = "application/json"
        encoded = json.dumps(body, separators=(",", ":")).encode("utf-8")
    if cart_token is not None:
        headers["Cart-Token"] = cart_token
    request = urllib.request.Request(client.base_url + path, data=encoded, headers=headers, method=method)
    try:
        response = client.opener.open(request, timeout=client.timeout)
    except urllib.error.HTTPError as error:
        response = error
    if 300 <= response.status < 400:
        location = response.headers.get("Location", "")
        resolved = urllib.parse.urljoin(client.base_url + path, location)
        if urllib.parse.urlsplit(resolved).netloc != urllib.parse.urlsplit(client.base_url).netloc:
            raise Blocked("Store API attempted an off-origin redirect.")
        raise Blocked("Store API endpoint redirected unexpectedly.")
    raw = response.read(client.max_body_bytes + 1)
    if len(raw) > client.max_body_bytes:
        raise Blocked("Store API response exceeded the evidence bound.")
    payload = load_unique_json(raw.decode("utf-8"))
    if not isinstance(payload, dict):
        raise Blocked("Store API response was not a JSON object.")
    return response.status, payload, response.headers.get("Cart-Token")
```
